import org.apache.poi.ss.usermodel.Cell;
import org.apache.poi.ss.usermodel.Row;

import org.apache.poi.xssf.usermodel.XSSFWorkbook;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;

public class Practice5 {

    public static void writeData(
            String filePath,
            String fileName,
            String sheetName,
            String[] data
    ) throws IOException {
        File file = new File(filePath + "\\" + fileName);

        XSSFWorkbook workbook;
        XSSFSheet sheet;

        if (!file.exists()) {
            workbook = new XSSFWorkbook();
            sheet = workbook.createSheet(sheetName);
        } else {
            FileInputStream fis = new FileInputStream(file);
            workbook = new XSSFWorkbook(fis);

            sheet = workbook.getSheet(sheetName);

            if (sheet == null) {
                sheet = workbook.createSheet(sheetName);
            }

            fis.close();
        }

        int rowNum = sheet.getPhysicalNumberOfRows();
        Row newRow = sheet.createRow(rowNum);

        for (int i = 0; i < data.length; i++) {
            Cell cell = newRow.createCell(i);
            cell.setCellValue(data[i]);
        }

        FileOutputStream fos = new FileOutputStream(file);
        workbook.write(fos);

        fos.close();
        workbook.close();
    }

        String[][] students = {

import org.apache.poi.ss.usermodel.*;
import org.apache.poi.xssf.usermodel.XSSFSheet;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.IOException;

public class Practice8 {

    public static void main(String[] args) throws IOException {
        String path = System.getProperty("user.dir")
                + "\\src\\main\\resources\\Practical6.xlsx";

        File file = new File(path);

        XSSFWorkbook workbook;
        XSSFSheet sheet;

        if (!file.exists()) {
            workbook = new XSSFWorkbook();
            sheet = workbook.createSheet("Student_Marks");

            Row header = sheet.createRow(0);

            header.createCell(0).setCellValue("Student Name");
            header.createCell(1).setCellValue("Java");
            header.createCell(2).setCellValue("Python");
            header.createCell(3).setCellValue("C++");

            Object[][] data = {
                    {"Student1", 75, 55, 45},
                    {"Student2", 40, 35, 50},
                    {"Student3", 55, 62, 58},
                    {"Student4", 80, 90, 85},
                    {"Student5", 45, 50, 40},
                    {"Student6", 65, 55, 59},
                    {"Student7", 30, 45, 50},
                    {"Student8", 70, 72, 68},
                    {"Student9", 58, 59, 61},
                    {"Student10", 35, 40, 42}
            };

            for (int i = 0; i < data.length; i++) {
                Row row = sheet.createRow(i + 1);

                for (int j = 0; j < data[i].length; j++) {
                    Cell cell = row.createCell(j);

                    if (data[i][j] instanceof String) {
                        cell.setCellValue((String) data[i][j]);
                    } else {
                        cell.setCellValue((Integer) data[i][j]);
                    }
                }
            }

            FileOutputStream fos = new FileOutputStream(file);
            workbook.write(fos);
            fos.close();
        } else {
            FileInputStream fis = new FileInputStream(file);
            workbook = new XSSFWorkbook(fis);
            sheet = workbook.getSheet("Student_Marks");
            fis.close();
        }

        int totalStudents = sheet.getLastRowNum();
        int count = 0;

        System.out.println(
                "Students Scoring Above 60 in At Least One Subject\n"
        );

        for (int i = 1; i <= totalStudents; i++) {
            boolean passed = false;

            for (int j = 1; j <= 3; j++) {
                double marks = sheet.getRow(i)
                        .getCell(j)
                        .getNumericCellValue();

                if (marks > 60) {
                    passed = true;
                    break;
                }
            }

            if (passed) {
                System.out.println(
                        sheet.getRow(i)
                                .getCell(0)
                                .getStringCellValue()
                );

                count++;
            }
        }

        double percentage = (count * 100.0) / totalStudents;

        System.out.println("\nTotal Students : " + totalStudents);
        System.out.println("Students Above 60 : " + count);
        System.out.printf("Percentage : %.2f%%", percentage);

        workbook.close();
    }
}

import org.openqa.selenium.By;

import java.time.Duration;
import java.util.ArrayList;
import java.util.Collections;

public class Pract_12 {

    public static void main(String[] args) {
        WebDriver driver = new ChromeDriver();

        try {
            driver.manage().window().maximize();
            driver.manage().timeouts()
                    .implicitlyWait(Duration.ofSeconds(10));

            driver.get(
                    "https://datatables.net/examples/basic_init/zero_configuration.html"
            );

            int rows = driver.findElements(
                    By.xpath("//tbody/tr")
            ).size();

            System.out.println("Total Rows : " + rows);

            int cols = driver.findElements(
                    By.xpath("//thead/tr/th")
            ).size();

            System.out.println("Total Columns : " + cols);
            System.out.println("\n----- Table Data -----");

            List<WebElement> tableRows =
                    driver.findElements(By.xpath("//tbody/tr"));

            for (WebElement row : tableRows) {
                System.out.println(row.getText());
            }

            System.out.println("\nClicking Name Column for Sorting...");

            driver.findElement(
                    By.xpath("//thead/tr/th[1]")
            ).click();

            Thread.sleep(2000);

            List<WebElement> names =
                    driver.findElements(By.xpath("//tbody/tr/td[1]"));

            List<String> actualList = new ArrayList<>();

            for (WebElement name : names) {
                actualList.add(name.getText());
            }

            List<String> ascendingList = new ArrayList<>(actualList);
            Collections.sort(ascendingList);

            List<String> descendingList = new ArrayList<>(ascendingList);
            Collections.reverse(descendingList);

            if (actualList.equals(ascendingList)) {
                System.out.println(
                        "Sorting Validation Passed - Ascending Order"
                );
            } else if (actualList.equals(descendingList)) {
                System.out.println(
                        "Sorting Validation Passed - Descending Order"
                );
            } else {
                System.out.println("Sorting Validation Failed");
            }

            System.out.println("\n----- Search / Filter Validation -----");

            WebElement searchBox = driver.findElement(
                    By.xpath("//input[@type='search']")
            );

            searchBox.clear();
            searchBox.sendKeys("Tiger Nixon");

            Thread.sleep(2000);

            String actualName = driver.findElement(
                    By.xpath("//tbody/tr/td[1]")
            ).getText();

            if (actualName.equals("Tiger Nixon")) {
                System.out.println("Filter Validation Passed");
            } else {
                System.out.println("Filter Validation Failed");
            }

            System.out.println("\nFiltered Record:");
            System.out.println(
                    driver.findElement(By.xpath("//tbody/tr")).getText()
            );
        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());
        } finally {
            driver.quit();
            System.out.println("\nBrowser Closed Successfully.");
        }
    }
}

    public static String runTest(WebDriver driver, String browser) {
        try {
            System.out.println("\nTesting on " + browser);
            driver.get("https://www.google.com");

                    driver, Duration.ofSeconds(10)
            );

            String title = driver.getTitle();
            System.out.println("Page Title: " + title);

                throw new Exception("Page title not available");

                    ExpectedConditions.visibilityOfElementLocated(
                            By.name("q")

            ).sendKeys("Selenium WebDriver");

            wait.until(
                    ExpectedConditions.titleContains("Selenium WebDriver")
            );

            return "PASS";

            System.out.println("Compatibility Issue: " + e.getMessage());

                chromeIssue = e.getMessage();

                firefoxIssue = e.getMessage();
            }

            if (browser.equals("Edge")) {
                edgeIssue = e.getMessage();

            cell.setCellValue(data[i]);

    static String gridURL = "http://localhost:4444";

    static String chromeStatus = "NOT RUN";
    static String firefoxStatus = "NOT RUN";
    static String edgeStatus = "NOT RUN";

    static String chromeIssue = "None";
    static String firefoxIssue = "None";
    static String edgeIssue = "None";

    static long chromeTime;
    static long firefoxTime;
    static long edgeTime;

    public static void main(String[] args) {
        testChrome();
        testFirefox();
        testEdge();

        displayComparison();
    }

    public static void testChrome() {
        long startTime = System.currentTimeMillis();
        WebDriver driver = null;

        try {
            ChromeOptions options = new ChromeOptions();

            driver = new RemoteWebDriver(
                    URI.create(gridURL).toURL(),
                    options
            );

            chromeStatus = runTest(driver, "Chrome");
        } catch (Exception e) {
            chromeStatus = "FAIL";
            chromeIssue = e.getMessage();

            System.out.println("Chrome Test Failed: " + e.getMessage());
        } finally {
            if (driver != null) {
                driver.quit();
            }

            chromeTime = System.currentTimeMillis() - startTime;

            System.out.println(
                    "Chrome Execution Time: " + chromeTime + " ms"
            );
        }
    }

    public static void testFirefox() {
        long startTime = System.currentTimeMillis();
        WebDriver driver = null;

        try {
            FirefoxOptions options = new FirefoxOptions();

            driver = new RemoteWebDriver(
                    URI.create(gridURL).toURL(),
                    options
            );

            firefoxStatus = runTest(driver, "Firefox");
        } catch (Exception e) {
            firefoxStatus = "FAIL";
            firefoxIssue = e.getMessage();

            System.out.println("Firefox Test Failed: " + e.getMessage());
        } finally {
            if (driver != null) {
                driver.quit();
            }

            firefoxTime = System.currentTimeMillis() - startTime;

            System.out.println(
                    "Firefox Execution Time: " + firefoxTime + " ms"
            );
        }
    }

    public static void testEdge() {
        long startTime = System.currentTimeMillis();
        WebDriver driver = null;

        try {
            EdgeOptions options = new EdgeOptions();

            driver = new RemoteWebDriver(
                    URI.create(gridURL).toURL(),
                    options
            );

            edgeStatus = runTest(driver, "Edge");
        } catch (Exception e) {
            edgeStatus = "FAIL";
            edgeIssue = e.getMessage();

            System.out.println("Edge Test Failed: " + e.getMessage());
        } finally {
            if (driver != null) {
                driver.quit();
            }

            edgeTime = System.currentTimeMillis() - startTime;

            System.out.println(
                    "Edge Execution Time: " + edgeTime + " ms"
            );
        }
    }

    public static String runTest(WebDriver driver, String browser) {
        try {
            System.out.println("\nTesting on " + browser);
            driver.get("https://www.google.com");

            WebDriverWait wait = new WebDriverWait(
                    driver, Duration.ofSeconds(10)
            );

            String title = driver.getTitle();
            System.out.println("Page Title: " + title);

            if (title == null || title.isEmpty()) {
                throw new Exception("Page title not available");
            }

            wait.until(
                    ExpectedConditions.visibilityOfElementLocated(
                            By.name("q")
                    )
            ).sendKeys("Selenium WebDriver");

            driver.findElement(By.name("q")).submit();

            wait.until(
                    ExpectedConditions.titleContains("Selenium WebDriver")
            );

            System.out.println(browser + " Test Passed");
            return "PASS";
        } catch (Exception e) {
            System.out.println(browser + " Test Failed");
            System.out.println("Compatibility Issue: " + e.getMessage());

            if (browser.equals("Chrome")) {
                chromeIssue = e.getMessage();
            }

            if (browser.equals("Firefox")) {
                firefoxIssue = e.getMessage();
            }

            if (browser.equals("Edge")) {
                edgeIssue = e.getMessage();
            }

            return "FAIL";
        }
    }

    public static void displayComparison() {
        System.out.println("\n");
        System.out.println(
                "=============================================================="
        );
        System.out.println(
                "              CROSS-BROWSER TESTING COMPARISON"
        );
        System.out.println(
                "=============================================================="
        );

        System.out.printf(
                "%-12s %-12s %-18s %-30s%n",
                "Browser", "Status", "Execution Time", "Compatibility Issue"
        );

        System.out.println(
                "--------------------------------------------------------------------------"
        );

        System.out.printf(
                "%-12s %-12s %-18s %-30s%n",
                "Chrome", chromeStatus, chromeTime + " ms", chromeIssue
        );
        System.out.printf(
                "%-12s %-12s %-18s %-30s%n",
                "Firefox", firefoxStatus, firefoxTime + " ms", firefoxIssue
        );
        System.out.printf(
                "%-12s %-12s %-18s %-30s%n",
                "Edge", edgeStatus, edgeTime + " ms", edgeIssue
        );

        System.out.println(
                "--------------------------------------------------------------------------"
        );

        long fastestTime = Long.MAX_VALUE;
        String fastestBrowser = "None";

        if (chromeStatus.equals("PASS") && chromeTime < fastestTime) {
            fastestTime = chromeTime;
            fastestBrowser = "Chrome";
        }

        if (firefoxStatus.equals("PASS") && firefoxTime < fastestTime) {
            fastestTime = firefoxTime;
            fastestBrowser = "Firefox";
        }

        if (edgeStatus.equals("PASS") && edgeTime < fastestTime) {
            fastestTime = edgeTime;
            fastestBrowser = "Edge";
        }

        if (fastestBrowser.equals("None")) {
            System.out.println("No browser passed the test.");
        } else {
            System.out.println(
                    "Fastest Browser: " + fastestBrowser
                            + " (" + fastestTime + " ms)"
            );
        }

        System.out.println(
                "=============================================================="
        );
    }
}
